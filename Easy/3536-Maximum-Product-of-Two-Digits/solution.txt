class Solution(object):
    def maxProduct(self, n):
        """
        :type n: int
        :rtype: int
        """
        n = str(n)
        a, b = 0, 0

        for char in n :
            new = int(char)
            if new >= a :
                b = a
                a = new
            elif new > b :
                b = new
        return a*b
