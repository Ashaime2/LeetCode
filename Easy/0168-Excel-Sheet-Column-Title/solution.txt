class Solution(object):
    def convertToTitle(self, columnNumber):
        """
        :type columnNumber: int
        :rtype: str
        """
        dico = {i: chr(65 + i) for i in range(26)}
        string = ''

        while columnNumber > 0:
            columnNumber -= 1
            reste = columnNumber%26
            string = dico[reste] + string
            columnNumber //=  26
        
        return string
