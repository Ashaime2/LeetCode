class Solution(object):
    def minimumPushes(self, word):
        """
        :type word: str
        :rtype: int
        """
        n = len(word)
        if n <= 8 : 
            return n
        if 8 < n <= 16 :
            return (n%8)*2 + 8
        if 16 < n <= 24 :
            return (n%16)*3 + ((n-16)%8)*2 + 8
        return (n%24)*4 + (n%16)*3 + ((n-16)%8)*2 + 8
