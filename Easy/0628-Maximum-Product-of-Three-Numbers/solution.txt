class Solution(object):
    def maximumProduct(self, nums):
        max1 = max2 = max3 = float("-inf")
        min1 = min2 = float("inf")

        for x in nums:
            # 3 plus grands
            if x >= max1:
                max1, max2, max3 = x, max1, max2
            elif x >= max2:
                max2, max3 = x, max2
            elif x >= max3:
                max3 = x

            # 2 plus petits
            if x <= min1:
                min1, min2 = x, min1
            elif x <= min2:
                min2 = x

        return max(max1 * max2 * max3, max1 * min1 * min2)
